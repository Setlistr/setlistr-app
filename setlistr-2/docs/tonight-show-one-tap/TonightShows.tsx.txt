'use client'
// PROPOSED (prototype): "Tonight's show, one tap". One shared vocabulary for
// the Dashboard row and the gold capture button's first screen, so both say
// and do the same thing. Every Start goes through the existing linked start
// (useShowActions → POST /api/shows/schedule/[id]/start → live, autostart).
import { ChevronRight, Mic, Play } from 'lucide-react'
import type { TonightShow } from '@/lib/tonightShows'
import type { ScheduledShow } from './UpcomingShows'

type Item = TonightShow<ScheduledShow>
const SANS = '"DM Sans", system-ui, sans-serif'

// Same values as the Dashboard's existing bands (app/app/dashboard/page.tsx).
const band: React.CSSProperties = {
  width: '100%', display: 'flex', alignItems: 'center', gap: 12, minHeight: 'var(--d-band, 64px)', padding: '10px 12px',
  borderRadius: 18, border: 'none', cursor: 'pointer', textAlign: 'left', fontFamily: SANS,
  background: 'var(--sl-band-bg)', boxShadow: 'var(--sl-band-ring)', color: 'var(--sl-text)',
}
const title: React.CSSProperties = { display: 'block', fontSize: 16, fontWeight: 700, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }
const sub: React.CSSProperties = { display: 'block', fontSize: 13, color: 'var(--sl-text-2)', marginTop: 2 }
const chevron = <ChevronRight size={18} aria-hidden="true" style={{ color: 'var(--sl-text-3)', flexShrink: 0 }} />

function StatusLine({ color, children }: { color: string; children: React.ReactNode }) {
  return (
    <span style={{ display: 'flex', alignItems: 'center', gap: 6, fontSize: 12, fontWeight: 700, color, marginBottom: 1 }}>
      <span aria-hidden="true" style={{ width: 7, height: 7, borderRadius: '50%', background: color, flexShrink: 0 }} />
      {children}
    </span>
  )
}

export function scheduledLabel(items: Item[]): string {
  const day = items[0]?.dayLabel || 'Today'
  return `Scheduled · ${day}`
}

// ── Dashboard row ─────────────────────────────────────────────────────────
// Shown only to people who can capture, and only when no capture is in
// progress (the in-progress row always comes first, as today).
export function TonightBand({ items, timeText, onStart, onChoose, onAddTime }: {
  items: Item[]
  timeText: (i: Item) => string
  onStart: (i: Item) => void      // one startable show: opens the existing Start confirmation
  onChoose: () => void            // several shows: the gold button's choice screen
  onAddTime: (i: Item) => void    // date-only booking: Schedule's "Add a start time"
}) {
  if (!items.length) return null
  if (items.length > 1) {
    return (
      <button onClick={onChoose} aria-label={`Choose one of ${items.length} shows scheduled today`} style={band}>
        <span style={{ flex: 1, minWidth: 0 }}>
          <StatusLine color="var(--sl-ok)">{scheduledLabel(items)}</StatusLine>
          <span style={title}>{items.length} shows · choose one to start</span>
          <span style={{ ...sub, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>
            {items.map(i => `${i.show.venues?.name || 'Venue TBD'} ${i.kind === 'startable' ? timeText(i) : '(time not set)'}`).join(' · ')}
          </span>
        </span>
        {chevron}
      </button>
    )
  }
  const i = items[0]
  if (i.kind === 'needs_time') {
    return (
      <button onClick={() => onAddTime(i)} aria-label={`Add a start time for ${i.show.venues?.name || 'this show'}`} style={band}>
        <span style={{ flex: 1, minWidth: 0 }}>
          <StatusLine color="var(--sl-gold-text)">{scheduledLabel(items)}</StatusLine>
          <span style={title}>{i.show.venues?.name || 'Venue TBD'}</span>
          <span style={sub}>Add a start time to start capture</span>
        </span>
        {chevron}
      </button>
    )
  }
  return (
    <button onClick={() => onStart(i)} aria-label={`Start capture at ${i.show.venues?.name || 'this show'}, ${timeText(i)}`} style={band}>
      <span style={{ flex: 1, minWidth: 0 }}>
        <StatusLine color="var(--sl-ok)">{scheduledLabel(items)}</StatusLine>
        <span style={title}>{i.show.venues?.name || 'Venue TBD'}</span>
        <span style={sub}>{timeText(i)}{i.show.venues?.city ? ` · ${i.show.venues.city}` : ''} · Start capture</span>
      </span>
      {chevron}
    </button>
  )
}

// ── Gold capture button: first screen ─────────────────────────────────────
// Leads with the scheduled show(s); "A different show" opens today's
// unchanged choice (Start Live Capture / Upload a Recording).
export function TonightLead({ items, artistName, timeText, startingId, error, active, onStart, onAddTime, onRejoin, onDifferent }: {
  items: Item[]
  artistName: string
  timeText: (i: Item) => string
  startingId: string | null
  error: string
  active: { venue: string; startedMinutesAgo: number } | null
  onStart: (i: Item) => void
  onAddTime: (i: Item) => void
  onRejoin: () => void
  onDifferent: () => void
}) {
  return (
    <div style={{ animation: 'fadeUp 0.3s ease' }}>
      <h1 style={{ fontSize: 30, fontWeight: 800, color: 'var(--sl-text)', margin: '0 0 8px', letterSpacing: '-0.025em' }}>Where tonight?</h1>
      <p style={{ fontSize: 14, color: 'var(--sl-text-2)', margin: '0 0 20px' }}>
        {active ? `A show is already being captured for ${artistName}.` : items.length ? `Scheduled for ${artistName}${items.length > 1 ? ' — choose the show you’re at.' : '.'}` : `Nothing else is scheduled for ${artistName} right now.`}
      </p>

      {active ? (
        // Rejoin priority: an in-progress capture comes first; no new Start is offered.
        <button onClick={onRejoin} style={{ ...band, marginBottom: 12 }} aria-label={`Continue capture at ${active.venue}`}>
          <span aria-hidden="true" style={{ width: 40, height: 40, borderRadius: '50%', flexShrink: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', background: 'color-mix(in srgb, var(--sl-ok) 14%, transparent)', color: 'var(--sl-ok)' }}><Mic size={16} strokeWidth={2.4} /></span>
          <span style={{ flex: 1, minWidth: 0 }}>
            <StatusLine color="var(--sl-ok)">Show in progress</StatusLine>
            <span style={title}>{active.venue}</span>
            <span style={sub}>Tap to continue capture</span>
          </span>
          {chevron}
        </button>
      ) : items.map(i => (
        <div key={i.show.id} style={{ background: 'var(--sl-card-bg)', boxShadow: 'var(--sl-card-inset)', border: '1px solid color-mix(in srgb, var(--sl-accent) 30%, transparent)', borderRadius: 16, padding: '16px 16px 14px', marginBottom: 10 }}>
          <StatusLine color={i.kind === 'startable' ? 'var(--sl-ok)' : 'var(--sl-gold-text)'}>{`Scheduled · ${i.dayLabel}`}</StatusLine>
          <p style={{ fontSize: 20, fontWeight: 800, color: 'var(--sl-text)', margin: '2px 0 2px', letterSpacing: '-0.01em' }}>{i.show.venues?.name || 'Venue TBD'}</p>
          <p style={{ fontSize: 13, color: 'var(--sl-text-2)', margin: '0 0 2px' }}>
            {timeText(i)}{i.show.venues?.city ? ` · ${i.show.venues.city}` : ''}
          </p>
          <p style={{ fontSize: 12, color: 'var(--sl-text-3)', margin: '0 0 12px' }}>
            {i.show.import_batch_id ? `Imported from ${i.show.show_import_batches?.file_name || 'a spreadsheet'}` : i.show.scheduled_by_name ? `Added by ${i.show.scheduled_by_name}` : ''}
          </p>
          {i.kind === 'startable' ? (
            <button onClick={() => onStart(i)} disabled={startingId === i.show.id}
              style={{ width: '100%', minHeight: 48, display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8, background: 'var(--sl-accent)', border: 'none', borderRadius: 12, color: 'var(--sl-on-accent)', fontSize: 15, fontWeight: 800, cursor: 'pointer', fontFamily: 'inherit', opacity: startingId === i.show.id ? 0.7 : 1 }}>
              <Play size={14} /> {startingId === i.show.id ? 'Starting…' : 'Start capture'}
            </button>
          ) : (
            <button onClick={() => onAddTime(i)}
              style={{ width: '100%', minHeight: 48, background: 'transparent', border: '1px solid color-mix(in srgb, var(--sl-accent) 30%, transparent)', borderRadius: 12, color: 'var(--sl-gold-text)', fontSize: 14, fontWeight: 700, cursor: 'pointer', fontFamily: 'inherit' }}>
              Add a start time
            </button>
          )}
        </div>
      ))}

      {error && <p role="alert" style={{ fontSize: 13, color: 'var(--sl-danger)', margin: '4px 0 12px', lineHeight: 1.4 }}>{error}</p>}

      <button onClick={onDifferent}
        style={{ width: '100%', minHeight: 48, marginTop: 6, background: 'transparent', border: '1px solid color-mix(in srgb, var(--sl-text) 12%, transparent)', borderRadius: 14, color: 'var(--sl-text-2)', fontSize: 14, fontWeight: 700, cursor: 'pointer', fontFamily: 'inherit' }}>
        A different show
      </button>
    </div>
  )
}
